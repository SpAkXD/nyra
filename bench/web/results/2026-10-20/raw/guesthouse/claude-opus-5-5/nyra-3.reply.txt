enum Msg { Show(str), Toggle(int) }

var roomKind = "All"
var openFaq = -1

fn update(m: Msg) {
    match m {
        Msg.Show(k) => roomKind = k
        Msg.Toggle(i) => openFaq = openFaq == i ? -1 : i
    }
}

fn roomList() -> Node = stack {
    row {
        for k in ["All", "Double", "Family", "Suite"] { button(k, on: Msg.Show(k), active: k == roomKind) }
    }
    grid(3, anim: "stagger") {
        for r in roomData.filter(x => roomKind == "All" || x.kind == roomKind) {
            card(r.name, r.text, tag: r.kind, items: [r.size, r.guests, r.price], hover: "zoom") {
                img(r.image, r.alt, ratio: "4/3")
            }
        }
    }
}

fn faqList() -> Node = stack(anim: "stagger") {
    for i, q in faqs {
        stack {
            button(q.0, on: Msg.Toggle(i), style: "ghost", active: openFaq == i, icon: openFaq == i ? "minus" : "plus")
            if openFaq == i { p(q.1) }
        }
    }
}
