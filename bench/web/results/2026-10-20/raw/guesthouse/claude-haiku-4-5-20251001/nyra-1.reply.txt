enum Msg { FilterRooms(str), ToggleFaq(int), OpenLightbox(int) }
var roomFilter = "All"
var faqOpen: [bool] = [false, false, false, false]
var lightboxOpen = false
var lightboxIndex = 0

fn update(m: Msg) {
    match m {
        Msg.FilterRooms(t) => roomFilter = t,
        Msg.ToggleFaq(i) => faqOpen[i] = !faqOpen[i],
        Msg.OpenLightbox(i) => {
            lightboxOpen = true
            lightboxIndex = i
        }
    }
}

fn isReceptionOpen() -> bool {
    let now = time.now_ms()
    let hours = now / 3600000 % 24
    let day = (now / 86400000 + 4) % 7
    if day >= 5 {
        return hours >= 8 && hours < 20
    }
    return hours >= 7 && hours < 21
}
