enum Msg { FilterRooms(str), ToggleFaq(int) }
var roomFilter = "All"
var faqOpen: [bool] = [false, false, false, false]

fn update(m: Msg) {
  match m {
    Msg.FilterRooms(t) => roomFilter = t
    Msg.ToggleFaq(i) => faqOpen[i] = !faqOpen[i]
  }
}

fn isReceptionOpen() -> bool {
  let now = time.now_ms()
  let hour = (now / 1000 / 3600 + 1) % 24
  let day = (now / 1000 / 86400) % 7
  if day >= 5 {
    return hour >= 8 && hour < 20
  } else {
    return hour >= 7 && hour < 21
  }
}

fn badge() -> Node = chip(
  title: if isReceptionOpen() { "Reception open" } else { "Reception closed" },
  text: ""
)
