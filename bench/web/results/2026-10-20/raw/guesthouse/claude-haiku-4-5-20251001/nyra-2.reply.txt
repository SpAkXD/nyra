enum Msg { FilterRooms(str), ShowFaq(int) }
var roomFilter = "All"
var faqOpen: [bool] = [false, false, false, false]

fn update(m: Msg) {
    match m {
        Msg.FilterRooms(t) => roomFilter = t,
        Msg.ShowFaq(i) => {
            faqOpen[i] = !faqOpen[i]
        }
    }
}

fn receptionStatus() -> str {
    let now = time.now_ms()
    let date = time.now_ms()
    let h = (now / 3600000 + 1) % 24
    let d = (now / 86400000) % 7
    
    let isWeekday = d >= 1 && d <= 5
    let isSat = d == 6
    let isSun = d == 0
    
    if isWeekday {
        return h >= 7 && h < 21 ? "Reception open" : "Reception closed"
    } else if isSat || isSun {
        return h >= 8 && h < 20 ? "Reception open" : "Reception closed"
    }
    return "Reception closed"
}
