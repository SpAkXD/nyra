enum Msg { FilterRooms(str), OpenGallery(int), CloseGallery, NextGallery, PrevGallery, ToggleFaq(int) }

var roomFilter = "All"
var galleryOpen = -1
var faqOpen: [bool] = [false, false, false, false]

fn update(m: Msg) {
  match m {
    Msg.FilterRooms(f) => roomFilter = f,
    Msg.OpenGallery(i) => galleryOpen = i,
    Msg.CloseGallery => galleryOpen = -1,
    Msg.NextGallery => galleryOpen = (galleryOpen + 1) % 6,
    Msg.PrevGallery => galleryOpen = if galleryOpen == 0 { 5 } else { galleryOpen - 1 },
    Msg.ToggleFaq(i) => faqOpen[i] = !faqOpen[i],
  }
}

fn isReceptionOpen() -> bool {
  let now = time.now_ms()
  let d = (now / 1000) / 86400
  let sec = (now / 1000) % 86400
  let hour = sec / 3600
  let min = (sec % 3600) / 60
  let dow = (d + 4) % 7
  let openHour = if dow >= 5 { 8 } else { 7 }
  let closeHour = if dow >= 5 { 20 } else { 21 }
  let closeMins = 0
  return hour > openHour || (hour == openHour && min >= 0) && (hour < closeHour || (hour == closeHour && min < closeMins))
}

fn renderRooms() -> Node = stack {
  row {
    for f in ["All", "Double", "Family", "Suite"] {
      button(f, on: Msg.FilterRooms(f), active: f == roomFilter, style: if f == roomFilter { "primary" } else { "outline" })
    }
  }
  grid(3, anim: "stagger", hover: "lift") {
    for r in roomData.filter(x => roomFilter == "All" || x.1 == roomFilter) {
      card(r.0, r.5, image: r.6) {
        p("{r.1} · {r.2} m² · {r.3} guests")
        p("{r.4} Kč per night")
      }
    }
  }
}

fn renderGallery() -> Node = grid(3) {
  for i, g in galleryData {
    img(g.0, g.1, ratio: "4/3", href: "", on: Msg.OpenGallery(i))
  }
}

fn renderFaq() -> Node = stack {
  for i, fq in faqData {
    let isOpen = faqOpen[i]
    stack {
      button(fq.0, on: Msg.ToggleFaq(i), style: "ghost", icon: if isOpen { ":chevron-up:" } else { ":chevron-down:" })
      if isOpen {
        p(fq.1)
      }
    }
  }
}
