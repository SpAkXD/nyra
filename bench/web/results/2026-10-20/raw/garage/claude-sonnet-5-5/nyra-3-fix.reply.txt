fn update(m: Msg) {
    match m {
        Msg.Tab(k) => tab = k
        Msg.Size(k) => size = k
        Msg.Toggle(k) => {
            if picked.contains(k) {
                picked = picked.filter(x => x != k)
            } else {
                picked.push(k)
            }
        }
    }
}

fn money(n: int) -> str {
    let s = str(n)
    if n >= 1000 {
        return s.slice(0, s.len() - 3) + "," + s.slice(s.len() - 3, s.len()) + " Kč"
    }
    return s + " Kč"
}

fn idx() -> int = size == "Small car" ? 0 : (size == "Family car" ? 1 : 2)

fn total() -> int {
    var sum = 0
    for s in svc {
        if picked.contains(s.0) { sum += s.1[idx()] }
    }
    return sum
}

fn servicesTabs() -> Node = stack {
    row {
        for t in tabs { button(t.0, on: Msg.Tab(t.0), active: t.0 == tab) }
    }
    for t in tabs.filter(x => x.0 == tab) {
        card(t.0, t.1, items: t.2)
    }
}

fn calculator() -> Node = stack {
    row {
        for s in sizes { button(s, on: Msg.Size(s), active: s == size) }
    }
    grid(1) {
        for s in svc {
            button(s.0 + " – " + money(s.1[idx()]), on: Msg.Toggle(s.0), active: picked.contains(s.0), icon: picked.contains(s.0) ? "check" : "plus")
        }
    }
    card("Estimated total", money(total()), icon: "check")
    button("Book this service", "#contact", style: "accent")
}
