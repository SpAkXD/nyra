use web

site("Autoservis Dvořák", primary: "#1F4E8C", accent: "#F2A900", bg: "#F4F5F7", font: "Source Sans 3", display: "Barlow Condensed", radius: 12)

enum Msg { SelectService(str), SelectCarSize(str), ToggleServiceCheckbox(str) }

var selectedService = "Servicing"
var selectedCarSize = "Family car"
var checkedServices: [str: bool] = [:]

fn update(m: Msg) {
    match m {
        Msg.SelectService(s) => selectedService = s,
        Msg.SelectCarSize(size) => selectedCarSize = size,
        Msg.ToggleServiceCheckbox(s) => {
            let current = checkedServices.get(s, false)
            checkedServices[s] = !current
        }
    }
}

fn getPrice(service: str, carSize: str) -> int {
    let prices: [str: [str: int]] = [
        "Oil service": ["Small car": 1490, "Family car": 1790, "SUV / van": 2290],
        "Front brake pads": ["Small car": 2190, "Family car": 2590, "SUV / van": 3190],
        "Air-con service": ["Small car": 1290, "Family car": 1290, "SUV / van": 1590],
        "Wheel alignment": ["Small car": 990, "Family car": 990, "SUV / van": 1290],
        "Tyre change (4 wheels)": ["Small car": 800, "Family car": 960, "SUV / van": 1200]
    ]
    return prices.get(service, [:]).get(carSize, 0)
}

fn calculateTotal() -> int {
    var total = 0
    for service in ["Oil service", "Front brake pads", "Air-con service", "Wheel alignment", "Tyre change (4 wheels)"] {
        if checkedServices.get(service, false) {
            total += getPrice(service, selectedCarSize)
        }
    }
    return total
}

        button("Bodywork", on: Msg.SelectService("Bodywork"), active: selectedService == "Bodywork", style: if selectedService == "Bodywork" { "primary" } else { "ghost" })
    }
    if selectedService == "Servicing" {
        p("Scheduled servicing to the maker's plan, so your warranty stays valid.")

        p("Seasonal tyre changes, storage and repairs while you wait.")
        list(["Tyre fitting and balancing", "Tyre hotel for 1,200 Kč a season", "Puncture repair"])

        p("We read the fault codes and find the real cause before we replace anything.")
        list(["Engine and electronics diagnostics", "Air-con check and refill", "Pre-purchase inspection"])

    if selectedService == "Bodywork" {
        p("Dents, scratches and paint, matched in our own paint shop.")
        list(["Dent repair", "Colour-matched paintwork", "Insurance claims handled for you"])
    }
}

fn renderPriceCalculator() -> Node = stack {
    h3("Select car size")
    row {
        button("Small car", on: Msg.SelectCarSize("Small car"), active: selectedCarSize == "Small car", style: if selectedCarSize == "Small car" { "primary" } else { "ghost" })
        button("Family car", on: Msg.SelectCarSize("Family car"), active: selectedCarSize == "Family car", style: if selectedCarSize == "Family car" { "primary" } else { "ghost" })
        button("SUV / van", on: Msg.SelectCarSize("SUV / van"), active: selectedCarSize == "SUV / van", style: if selectedCarSize == "SUV / van" { "primary" } else { "ghost" })
    }
    h3("Select services")
    for service in ["Oil service", "Front brake pads", "Air-con service", "Wheel alignment", "Tyre change (4 wheels)"] {
        let price = getPrice(service, selectedCarSize)
        row {
            field(service, service, type: "check", req: false)
            p("{price} Kč")
        }
    }
    let total = calculateTotal()
    h3("Estimated total")
    p(style: "primary" cls: "text-2xl font-bold") {
        "{total} Kč"
    }
    button("Book this service", href: "#contact", style: "primary")
}
