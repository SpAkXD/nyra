        // Mobile menu toggle
        const menuToggle = document.getElementById('menuToggle');

        navLinks.querySelectorAll('a').forEach(link => {
            link.addEventListener('click', function() {
                navLinks.classList.remove('mobile-open');
            });
        });

        // Services tab switching
        const tabButtons = document.querySelectorAll('.tab-button');
        const tabPanels = document.querySelectorAll('.tab-panel');

        tabButtons.forEach(button => {
            button.addEventListener('click', function() {
                const tabId = this.getAttribute('data-tab');
                
                tabButtons.forEach(btn => btn.classList.remove('active'));
                tabPanels.forEach(panel => panel.classList.remove('active'));
                
                this.classList.add('active');
                document.getElementById(tabId).classList.add('active');
            });
        });

        // Price calculator
        let currentSize = 'family';
        let selectedServices = {};

        const sizeButtons = document.querySelectorAll('.size-button');
        const serviceCheckboxes = document.querySelectorAll('.service-item input[type="checkbox"]');
        const totalPriceElement = document.getElementById('totalPrice');

        sizeButtons.forEach(button => {
            button.addEventListener('click', function() {
                sizeButtons.forEach(btn => btn.classList.remove('active'));
                this.classList.add('active');
                currentSize = this.getAttribute('data-size');
                updatePrices();
                updateTotal();
            });
        });

        serviceCheckboxes.forEach(checkbox => {
            checkbox.addEventListener('change', function() {
                if (this.checked) {
                    selectedServices[this.id] = true;
                } else {
                    delete selectedServices[this.id];
                }
                updateTotal();
            });
        });

        function updatePrices() {
            serviceCheckboxes.forEach(checkbox => {
                const price = checkbox.getAttribute(`data-${currentSize}`);
                const label = checkbox.nextElementSibling;
                const priceSpan = label.querySelector('.service-price');
                priceSpan.textContent = new Intl.NumberFormat('cs-CZ').format(price) + ' Kč';
            });
        }

        function updateTotal() {
            let total = 0;
            Object.keys(selectedServices).forEach(serviceId => {
                const checkbox = document.getElementById(serviceId);
                const price = parseInt(checkbox.getAttribute(`data-${currentSize}`));
                total += price;
            });
            totalPriceElement.textContent = new Intl.NumberFormat('cs-CZ').format(total) + ' Kč';
        }

        // Number counter animation
        function animateNumbers() {
            const numberElements = document.querySelectorAll('.number-value[data-count]');
            const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

            numberElements.forEach(element => {
                const target = parseInt(element.getAttribute('data-count'));
                
                if (prefersReducedMotion) {
                    element.textContent = target.toLocaleString('cs-CZ');
                } else {
                    const duration = 2000;
                    const start = Date.now();
                    
                    function update() {
                        const now = Date.now();
                        const progress = Math.min((now - start) / duration, 1);
                        const current = Math.floor(progress * target);
                        element.textContent = current.toLocaleString('cs-CZ');
                        
                        if (progress < 1) {
                            requestAnimationFrame(update);
                        }
                    }
                    
                    requestAnimationFrame(update);
                }
            });
        }

        // Intersection Observer for animations
        const observerOptions = {
            threshold: 0.1,
            rootMargin: '0px 0px -100px 0px'
        };

        const observer = new IntersectionObserver(function(entries) {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    if (entry.target.classList.contains('numbers-section')) {
                        animateNumbers();
                        observer.unobserve(entry.target);
                    }
                }
            });
        }, observerOptions);

        const numbersSection = document.querySelector('.numbers-section');
        if (numbersSection) {
            observer.observe(numbersSection);
        }

        // Contact form submission
        const contactForm = document.getElementById('contactForm');
        contactForm.addEventListener('submit', function(e) {
            e.preventDefault();
            
            const name = document.getElementById('name').value;
            const phone = document.getElementById('phone').value;
            const email = document.getElementById('email').value;
            const car = document.getElementById('car').value;
            const service = document.getElementById('service').value;
            const date = document.getElementById('date').value;
            const message = document.getElementById('message').value;
            
            const body = `Booking details:\n\nName: ${name}\nPhone: ${phone}\nE-mail: ${email}\nCar: ${car}\nService: ${service}\nPreferred date: ${date}\nMessage: ${message}`;
            
            const mailtoLink = `mailto:servis@autodvorak.cz?subject=Service booking&body=${encodeURIComponent(body)}`;
            window.location.href = mailtoLink;
        });
    </script>
