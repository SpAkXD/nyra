        // Menu toggle
        const menuToggle = document.getElementById('menuToggle');
        const navMenu = document.getElementById('navMenu');

        menuToggle.addEventListener('click', () => {
            navMenu.classList.toggle('active');
        });

        // Close menu when a link is clicked
        navMenu.querySelectorAll('a').forEach(link => {
            link.addEventListener('click', () => {
                navMenu.classList.remove('active');
            });
        });

        // Services tabs
        const serviceTabs = document.querySelectorAll('.services-tab');
        const servicePanels = document.querySelectorAll('.services-panel');

        serviceTabs.forEach(tab => {
            tab.addEventListener('click', () => {
                const tabName = tab.getAttribute('data-tab');
                
                serviceTabs.forEach(t => t.classList.remove('active'));
                servicePanels.forEach(p => p.classList.remove('active'));
                
                tab.classList.add('active');
                document.querySelector(`[data-panel="${tabName}"]`).classList.add('active');
            });
        });

        // Price calculator
        const carSizeButtons = document.querySelectorAll('.car-size-btn');
        const priceCheckboxes = document.querySelectorAll('.price-items input[type="checkbox"]');
        const priceTotal = document.getElementById('priceTotal');
        let currentSize = 'family';

        carSizeButtons.forEach(btn => {
            btn.addEventListener('click', () => {
                carSizeButtons.forEach(b => b.classList.remove('active'));
                btn.classList.add('active');
                currentSize = btn.getAttribute('data-size');
                
                // Update displayed prices
                priceCheckboxes.forEach(checkbox => {
                    const parent = checkbox.closest('.price-item');
                    const priceDisplay = parent.querySelector('.price-item-price');
                    const price = checkbox.getAttribute(`data-${currentSize}`);
                    priceDisplay.textContent = formatPrice(price);
                });
                
                updateTotal();
            });
        });

        priceCheckboxes.forEach(checkbox => {
            checkbox.addEventListener('change', updateTotal);
        });

        function formatPrice(price) {
            return price.replace(/\B(?=(\d{3})+(?!\d))/g, ',') + ' Kč';
        }

        function updateTotal() {
            let total = 0;
            priceCheckboxes.forEach(checkbox => {
                if (checkbox.checked) {
                    const price = parseInt(checkbox.getAttribute(`data-${currentSize}`));
                    total += price;
                }
            });
            priceTotal.textContent = formatPrice(total.toString());
        }

        // Number counter animation
        const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

        function countUp(element) {
            const target = parseInt(element.getAttribute('data-target'));
            if (isNaN(target)) return;

            if (prefersReducedMotion) {
                element.textContent = target.toLocaleString('cs-CZ');
                return;
            }

            let current = 0;
            const increment = Math.ceil(target / 30);
            const timer = setInterval(() => {
                current += increment;
                if (current >= target) {
                    current = target;
                    clearInterval(timer);
                }
                element.textContent = current.toLocaleString('cs-CZ');
            }, 30);
        }

        // Intersection Observer for animations
        const observerOptions = {
            threshold: 0.1,
            rootMargin: '0px 0px -50px 0px'
        };

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const numberElements = entry.target.querySelectorAll('[data-target]');
                    numberElements.forEach(element => {
                        if (!element.hasAttribute('data-counted')) {
                            element.setAttribute('data-counted', 'true');
                            countUp(element);
                        }
                    });
                    observer.unobserve(entry.target);
                }
            });
        }, observerOptions);

        const numbersSection = document.querySelector('.numbers');
        observer.observe(numbersSection);

        // Form submission
        const contactForm = document.getElementById('contactForm');
        contactForm.addEventListener('submit', (e) => {
            e.preventDefault();
            
            const name = document.getElementById('name').value;
            const phone = document.getElementById('phone').value;
            const email = document.getElementById('email').value;
            const car = document.getElementById('car').value;
            const service = document.getElementById('service').value;
            const date = document.getElementById('date').value;
            const message = document.getElementById('message').value;
            
            const subject = 'Service booking';
            const body = `Name: ${name}
Phone: ${phone}
Email: ${email}
Car: ${car}
Service: ${service}
Preferred date: ${date || 'Not specified'}
Message: ${message || 'None'}`;
            
            const mailtoLink = `mailto:servis@autodvorak.cz?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
            window.location.href = mailtoLink;
        });
