site("Autoservis Dvořák", primary: "#1F4E8C", accent: "#F2A900", bg: "#F4F5F7", font: "Source Sans 3", display: "Barlow Condensed", radius: 12)

enum Msg { SelectService(str), SelectCar(str), ToggleService(str) }

var selected_service = "Servicing"
var selected_car = "Family car"
var services_checked: [str: bool] = [:]

fn update(m: Msg) {
    match m {
        Msg.SelectService(s) => selected_service = s,
        Msg.SelectCar(c) => selected_car = c,
        Msg.ToggleService(s) => {
            var current = services_checked.get(s, false)
            services_checked[s] = !current
        }
    }
}

fn get_price(service: str, car: str) -> int {
    var prices: [str: [str: int]] = [:]
    prices["Oil service"] = ["Small car": 1490, "Family car": 1790, "SUV / van": 2290]
    prices["Front brake pads"] = ["Small car": 2190, "Family car": 2590, "SUV / van": 3190]
    prices["Air-con service"] = ["Small car": 1290, "Family car": 1290, "SUV / van": 1590]
    prices["Wheel alignment"] = ["Small car": 990, "Family car": 990, "SUV / van": 1290]
    prices["Tyre change (4 wheels)"] = ["Small car": 800, "Family car": 960, "SUV / van": 1200]
    if let car_prices = prices.get(service) {
        return car_prices.get(car, 0)
    }
    return 0
}

fn calc_total() -> int {
    var total = 0
    for service in ["Oil service", "Front brake pads", "Air-con service", "Wheel alignment", "Tyre change (4 wheels)"] {
        if services_checked.get(service, false) {
            total += get_price(service, selected_car)
        }
    }
    return total
}
