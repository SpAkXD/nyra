enum Msg { Tab(str), Size(int), Toggle(int) }

var tab = "Servicing"
var size = 1
var picked = [false, false, false, false, false]

fn update(m: Msg) {
    match m {
        Msg.Tab(t) => tab = t
        Msg.Size(i) => size = i
        Msg.Toggle(i) => picked[i] = !picked[i]
    }
}

fn pad3(n: int) -> str {
    if n < 10 {
        return "00{n}"
    }
    if n < 100 {
        return "0{n}"
    }
    return str(n)
}

fn thousands(n: int) -> str {
    if n < 1000 {
        return str(n)
    }
    return "{thousands(n / 1000)},{pad3(n % 1000)}"
}

fn kc(n: int) -> str = "{thousands(n)} Kč"

fn total() -> int {
    var sum = 0
    for i, p in pricelist {
        if picked[i] {
            sum += p.1[size]
        }
    }
    return sum
}

    for s in svc.filter(x => x.0 == tab) {
        card(s.0, s.1, icon: s.2, items: s.3, anim: "fade", hover: "lift")
    }
}

    row {

    }
}
