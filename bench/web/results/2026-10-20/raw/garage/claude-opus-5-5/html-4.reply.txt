(function(){

  var menuBtn = document.querySelector('.menu-btn');

  menuBtn.addEventListener('click', function(){
    var open = links.classList.toggle('open');
    menuBtn.setAttribute('aria-expanded', open ? 'true' : 'false');
  });
  links.addEventListener('click', function(e){ if (e.target.closest('a')) closeMenu(); });
  document.addEventListener('keydown', function(e){
    if (e.key === 'Escape' && links.classList.contains('open')) { closeMenu(); menuBtn.focus(); }
  });
  window.addEventListener('resize', function(){ if (window.innerWidth > 920) closeMenu(); });

  /* Tabs */
  var tabs = Array.prototype.slice.call(document.querySelectorAll('[role="tab"]'));
  function selectTab(tab, focus){
    tabs.forEach(function(t){
      var on = t === tab;
      t.setAttribute('aria-selected', on ? 'true' : 'false');
      t.tabIndex = on ? 0 : -1;
      var p = document.getElementById(t.getAttribute('aria-controls'));
      if (p) p.hidden = !on;
    });
    if (focus) tab.focus();
  }
  tabs.forEach(function(tab, i){
    tab.addEventListener('click', function(){ selectTab(tab, false); });
    tab.addEventListener('keydown', function(e){
      var idx = null;
      if (e.key === 'ArrowRight' || e.key === 'ArrowDown') idx = (i + 1) % tabs.length;
      else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') idx = (i - 1 + tabs.length) % tabs.length;
      else if (e.key === 'Home') idx = 0;
      else if (e.key === 'End') idx = tabs.length - 1;
      if (idx !== null){ e.preventDefault(); selectTab(tabs[idx], true); }
    });
  });

  /* Price calculator */
  var services = [
    {id:'oil', name:'Oil service', p:{small:1490, family:1790, suv:2290}},
    {id:'brakes', name:'Front brake pads', p:{small:2190, family:2590, suv:3190}},
    {id:'aircon', name:'Air-con service', p:{small:1290, family:1290, suv:1590}},
    {id:'align', name:'Wheel alignment', p:{small:990, family:990, suv:1290}},
    {id:'tyres', name:'Tyre change (4 wheels)', p:{small:800, family:960, suv:1200}}
  ];
  var sizeNames = {small:'Small car', family:'Family car', suv:'SUV / van'};
  function fmt(n){ return n.toLocaleString('en-US') + ' Kč'; }
  var list = document.getElementById('svc-list');
  var checkSvg = '<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3.2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="m5 12.5 4.5 4.5L19 7.5"/></svg>';
  services.forEach(function(s){
    var wrap = document.createElement('div');
    wrap.className = 'svc';
    wrap.innerHTML = '<input type="checkbox" id="svc-' + s.id + '" value="' + s.id + '">' +
      '<label for="svc-' + s.id + '"><span class="box">' + checkSvg + '</span>' +
      '<span class="svc-name">' + s.name + '</span>' +
      '<span class="svc-price" id="price-' + s.id + '">' + fmt(s.p.family) + '</span></label>';
    list.appendChild(wrap);
  });
  var calc = document.getElementById('calc');
  var totalEl = document.getElementById('total');
  var metaEl = document.getElementById('sum-meta');
  var sumList = document.getElementById('sum-list');
  function update(){
    var sizeInput = calc.querySelector('input[name="size"]:checked');
    var size = sizeInput ? sizeInput.value : 'family';
    var total = 0, count = 0, items = [];
    services.forEach(function(s){
      var price = s.p[size];
      document.getElementById('price-' + s.id).textContent = fmt(price);
      if (document.getElementById('svc-' + s.id).checked){
        total += price; count++;
        items.push('<li><span>' + s.name + '</span><span>' + fmt(price) + '</span></li>');
      }
    });
    totalEl.textContent = fmt(total);
    metaEl.textContent = sizeNames[size] + ' · ' + (count === 0 ? 'no services selected' : count + (count === 1 ? ' service' : ' services'));
    sumList.innerHTML = items.length ? items.join('') : '<li class="empty">Tick the services you need.</li>';
  }
  calc.addEventListener('change', update);
  calc.addEventListener('input', update);
  update();

  /* Reveal on scroll */
  document.querySelectorAll('[data-stagger]').forEach(function(group){
    Array.prototype.forEach.call(group.querySelectorAll('.reveal'), function(el, i){
      el.style.setProperty('--i', i);
    });
  });
  var revealEls = document.querySelectorAll('.reveal, .reveal-line');
  var counters = document.querySelectorAll('[data-count]');

  function countUp(el){
    var target = parseInt(el.getAttribute('data-count'), 10);
    var dur = 1700, start = null;
    function frame(ts){
      if (start === null) start = ts;
      var t = Math.min((ts - start) / dur, 1);
      var eased = 1 - Math.pow(1 - t, 3);
      el.textContent = Math.round(target * eased).toLocaleString('en-US');
      if (t < 1) requestAnimationFrame(frame);
      else el.textContent = target.toLocaleString('en-US');
    }
    requestAnimationFrame(frame);
  }

  if (reduce || !('IntersectionObserver' in window)){
    Array.prototype.forEach.call(revealEls, function(el){ el.classList.add('in'); });
    // numbers already show their final value in the markup
  } else {
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if (entry.isIntersecting){
          entry.target.classList.add('in');
          io.unobserve(entry.target);
        }
      });
    }, {threshold:0.12, rootMargin:'0px 0px -40px 0px'});
    Array.prototype.forEach.call(revealEls, function(el){ io.observe(el); });

    Array.prototype.forEach.call(counters, function(el){ el.textContent = '0'; });
    var co = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if (entry.isIntersecting){
          countUp(entry.target);
          co.unobserve(entry.target);
        }
      });
    }, {threshold:0.5});
    Array.prototype.forEach.call(counters, function(el){ co.observe(el); });
  }

  Array.prototype.forEach.call(form.querySelectorAll('input, select, textarea'), function(el){
    el.addEventListener('input', function(){
      if (el.checkValidity()){
        el.closest('.field').classList.remove('invalid');
        el.removeAttribute('aria-invalid');
      }
    });
  });

  form.addEventListener('submit', function(e){
    e.preventDefault();
    var firstInvalid = null;
    Array.prototype.forEach.call(form.querySelectorAll('input, select, textarea'), function(el){
      var ok = el.checkValidity();
      el.closest('.field').classList.toggle('invalid', !ok);
      if (!ok){ el.setAttribute('aria-invalid','true'); if (!firstInvalid) firstInvalid = el; }
      else el.removeAttribute('aria-invalid');
    });
    if (firstInvalid){
      status.textContent = 'Please fill in the required fields.';
      firstInvalid.focus();
      if (form.reportValidity) form.reportValidity();
      return;
    }
    var v = function(id){ return document.getElementById(id).value.trim(); };
    var lines = [

      'Phone: ' + v('f-phone'),
      'E-mail: ' + (v('f-email') || '-'),
      'Car make and model: ' + v('f-car'),
      'Service: ' + v('f-service'),
      'Preferred date: ' + (v('f-date') || '-'),
      'Message: ' + (v('f-msg') || '-')
    ];
    var href = 'mailto:servis@autodvorak.cz?subject=' + encodeURIComponent('Service booking') +
               '&body=' + encodeURIComponent(lines.join('\r\n'));
    status.textContent = 'Opening your e-mail app…';
    window.location.href = href;
  });
})();
