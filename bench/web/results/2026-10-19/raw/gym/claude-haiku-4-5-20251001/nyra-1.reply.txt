enum Msg {
  FilterDay(str),
  ToggleBilling(bool),
  OpenGallery(int),
  CloseGallery,
  NextGallery,
  PrevGallery
}

var activeDay = "All"
var billingMonthly = true
var galleryOpen = false
var galleryIndex = 0

fn update(m: Msg) {
  match m {
    Msg.FilterDay(d) => activeDay = d,
    Msg.ToggleBilling(b) => billingMonthly = b,
    Msg.OpenGallery(i) => { galleryOpen = true, galleryIndex = i },
    Msg.CloseGallery => galleryOpen = false,
    Msg.NextGallery => galleryIndex = (galleryIndex + 1) % 8,
    Msg.PrevGallery => galleryIndex = (galleryIndex + 7) % 8
  }
}
