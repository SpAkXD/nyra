    match m {
        Msg.FilterClasses(day) => classFilter = day
        Msg.ToggleMembership => membershipMonthly = !membershipMonthly
    }
}

fn classesSection() -> Node = stack {
    row {
        for day in ["All", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"] {
            button(day, on: Msg.FilterClasses(day), active: day == classFilter, style: if day == classFilter { "primary" } else { "outline" })
        }
    }
    grid(3, anim: "stagger", hover: "lift") {
        for c in classes {
            if classFilter == "All" || c.0 == classFilter {
                card(c.2, c.3, tag: c.4, items: [c.0 + " · " + c.1])
            }
        }
    }
}

fn membershipsection() -> Node = stack {
    row {
        button("Monthly", on: Msg.ToggleMembership, active: membershipMonthly, style: if membershipMonthly { "primary" } else { "outline" })
        button("Yearly", on: Msg.ToggleMembership, active: !membershipMonthly, style: if !membershipMonthly { "primary" } else { "outline" })
    }
    grid(3, hover: "lift") {
        card(
            "Basic",
            if membershipMonthly { "790 Kč/month" } else { "7,900 Kč/year" },
            items: ["Gym floor access", "2 classes a week", "App booking"],
            href: "#contact"

        card(

        )
    }
}
