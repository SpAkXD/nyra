enum Msg { Day(str), Billing(str) }
var day = "All"
var billing = "Monthly"

fn update(m: Msg) {
    match m {
        Msg.Day(d) => day = d
        Msg.Billing(b) => billing = b
    }
}

fn schedule() -> Node = stack {
    row {
        for d in ["All", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"] {
            button(d, on: Msg.Day(d), active: d == day, style: "outline")
        }
    }
    grid(3, anim: "stagger", hover: "lift") {
        for c in classes.filter(x => day == "All" || x.0 == day) {
            card("{c.2}", "{c.3} · {c.4}", tag: "{c.0} · {c.1}")
        }
    }
}
