  function applyStagger(container) {

      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          revealObserver.unobserve(entry.target);
        }
      });

  function formatNum(n) { return Math.round(n).toLocaleString('en-US'); }

  function animateCount(c) {
    var target = +c.dataset.count, duration = 1800, start = null;
    function step(ts) {
      if (start === null) start = ts;
      var p = Math.min((ts - start) / duration, 1);
      var eased = 1 - Math.pow(1 - p, 4);
      c.textContent = formatNum(target * eased);
      if (p < 1) requestAnimationFrame(step); else showFinal(c);
    }
    requestAnimationFrame(step);
  }

    counters.forEach(showFinal);

      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          animateCount(entry.target);
          countObserver.unobserve(entry.target);
        }
      });

  function validateField(f) {
    var err = document.getElementById(f.id + '-error');
    var value = f.value.trim();
    var msg = '';
    if (!value) msg = messages[f.id];
    else if (f.type === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) msg = 'Please enter a valid e-mail address, e.g. name@example.com.';
    err.textContent = msg;
    if (msg) f.setAttribute('aria-invalid', 'true'); else f.removeAttribute('aria-invalid');
    return !msg;
  }
  requiredFields.forEach(function (f) {
    f.addEventListener('blur', function () { if (f.value.trim() || f.hasAttribute('aria-invalid')) validateField(f); });
    f.addEventListener('input', function () { if (f.hasAttribute('aria-invalid')) validateField(f); });
    f.addEventListener('change', function () { if (f.hasAttribute('aria-invalid')) validateField(f); });
  });
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    var firstInvalid = null;
    requiredFields.forEach(function (f) {
      if (!validateField(f) && !firstInvalid) firstInvalid = f;

      return;

    window.location.href = url;
  });
