enum Msg { Day(str), Billing(bool) }

var day = "All"
var yearly = false

fn update(m: Msg) {
    match m {
        Msg.Day(d) => day = d
        Msg.Billing(y) => yearly = y
    }
}

fn schedule() -> Node = stack {
    row {
        for d in days { button(d, on: Msg.Day(d), active: d == day, style: "ghost") }
    }
    grid(3, anim: "stagger", hover: "lift") {
        for s in sessions.filter(x => day == "All" || x.day == day) {
            card(s.name, "{s.coach} · {s.level}", tag: "{s.day} · {s.time}", icon: "dumbbell", cls: "surface")
        }
    }
}

fn pricing() -> Node = stack {
    row {
        button("Monthly", on: Msg.Billing(false), active: !yearly, style: "ghost")
        button("Yearly", on: Msg.Billing(true), active: yearly, style: "ghost")
    }
    grid(3, anim: "stagger", hover: "lift") {
        for p in plans {
            let price = yearly ? "**{p.year} Kč** per year" : "**{p.month} Kč** per month"
            if p.popular {
                card(p.name, price, items: p.features, tag: "Most popular", fill: true) {
                    button("Join now", "#contact", style: "accent")
                }
            } else {
                card(p.name, price, items: p.features, cls: "surface") {
                    button("Join now", "#contact", style: "outline")
                }
            }
        }
    }
}
