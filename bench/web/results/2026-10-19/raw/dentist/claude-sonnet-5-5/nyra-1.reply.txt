var openFaq = -1
enum Msg { Toggle(int) }
fn update(m: Msg) {
    match m {
        Msg.Toggle(i) => openFaq = if openFaq == i { -1 } else { i }
    }
}

fn faqList() -> Node = stack {
    for i, f in faqs {
        card(f.0, if openFaq == i { f.1 } else { "" }, icon: openFaq == i ? "chevron-up" : "chevron-down", on: Msg.Toggle(i))
    }
}
