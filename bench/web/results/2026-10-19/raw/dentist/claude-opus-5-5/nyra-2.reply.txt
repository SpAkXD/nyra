enum Msg { Toggle(int) }
var opened = [false, false, false, false, false]

fn update(m: Msg) {
    match m {
        Msg.Toggle(i) => opened[i] = !opened[i]
    }
}

fn faq_items() -> Node = stack {
    for i, item in questions {

                p(item.a)
            }
        }
    }
}
