(function () {
  'use strict';

    toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
    toggle.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
  }
  toggle.addEventListener('click', function () { setMenu(!menu.classList.contains('open')); });
  menu.addEventListener('click', function (e) { if (e.target.closest('a')) setMenu(false); });
  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && menu.classList.contains('open')) { setMenu(false); toggle.focus(); }
  });
  window.addEventListener('resize', function () { if (window.innerWidth > 880) setMenu(false); });

  /* Opening hours */
  var HOURS = { 0: null, 1: [480, 1080], 2: [480, 1080], 3: [480, 1080], 4: [480, 1080], 5: [480, 840], 6: null };
  function updateOpen() {
    var now = new Date();
    var day = now.getDay();
    var mins = now.getHours() * 60 + now.getMinutes();
    var h = HOURS[day];
    var open = !!(h && mins >= h[0] && mins < h[1]);
    document.querySelectorAll('[data-open-badge]').forEach(function (b) {
      b.textContent = open ? 'Open now' : 'Closed now';
      b.classList.toggle('is-open', open);
    });
    document.querySelectorAll('.hours tr').forEach(function (row) {
      var isToday = Number(row.getAttribute('data-day')) === day;
      row.classList.toggle('is-today', isToday);
      var th = row.querySelector('th');
      var tag = th.querySelector('.today-tag');
      if (isToday && !tag) {
        tag = document.createElement('span');
        tag.className = 'today-tag';
        tag.textContent = 'Today';
        th.appendChild(tag);
      } else if (!isToday && tag) {
        tag.remove();
      }
    });
  }
  updateOpen();
  setInterval(updateOpen, 30000);

  /* Count up */
  var fmt = function (n) { return Math.round(n).toLocaleString('en-US'); };
  var counters = Array.prototype.slice.call(document.querySelectorAll('.count'));
  function showFinal(el) { el.textContent = fmt(Number(el.getAttribute('data-target'))); }
  function animateCount(el) {
    var target = Number(el.getAttribute('data-target'));
    if (reduced()) { showFinal(el); return; }
    var dur = 1800, start = null;
    function step(ts) {
      if (start === null) start = ts;
      var p = Math.min((ts - start) / dur, 1);
      var eased = 1 - Math.pow(1 - p, 3);
      el.textContent = fmt(target * eased);
      if (p < 1) requestAnimationFrame(step); else showFinal(el);
    }
    requestAnimationFrame(step);
  }

  /* Reveal on scroll */
  var revealEls = Array.prototype.slice.call(document.querySelectorAll('.reveal, .reveal-card'));
  if (!('IntersectionObserver' in window) || reduced()) {
    revealEls.forEach(function (el) { el.classList.add('is-visible'); });
    counters.forEach(showFinal);
  } else {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    revealEls.forEach(function (el) { io.observe(el); });

          cio.unobserve(entry.target);

  dateInput.min = today.getFullYear() + '-' + pad(today.getMonth() + 1) + '-' + pad(today.getDate());

    var fields = form.querySelectorAll('input, select, textarea');

      return;
    }
    var v = function (id) { return document.getElementById(id).value.trim(); };
    var dateVal = v('f-date');
    var dateText = dateVal;
    var parts = dateVal.split('-');
    if (parts.length === 3) dateText = Number(parts[2]) + '. ' + Number(parts[1]) + '. ' + parts[0];
