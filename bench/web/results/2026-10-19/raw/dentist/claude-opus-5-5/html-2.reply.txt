(function(){

  var reduced = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  /* Nav */
  var nav = document.querySelector('.nav');

  var links = document.getElementById('nav-links');
  function closeMenu(){ links.classList.remove('open'); toggle.setAttribute('aria-expanded','false'); }

    toggle.setAttribute('aria-expanded', open ? 'true' : 'false');
  });

    if (e.key === 'Escape' && links.classList.contains('open')) { closeMenu(); toggle.focus(); }
  });
  window.addEventListener('resize', function(){ if (window.innerWidth > 900) closeMenu(); });
  function onScroll(){ nav.classList.toggle('scrolled', window.scrollY > 10); }
  window.addEventListener('scroll', onScroll, {passive:true}); onScroll();

  /* Opening hours (index = Date.getDay(), 0 = Sunday) */
  var hours = [null,[8,18],[8,18],[8,18],[8,18],[8,14],null];
  function updateStatus(){
    var now = new Date(), d = now.getDay(), h = hours[d];
    var mins = now.getHours()*60 + now.getMinutes();
    var open = !!h && mins >= h[0]*60 && mins < h[1]*60;
    document.querySelectorAll('.js-status').forEach(function(el){
      el.textContent = open ? 'Open now' : 'Closed now';
      el.classList.toggle('is-open', open);
      el.classList.toggle('is-closed', !open);
    });
    document.querySelectorAll('.hours tr').forEach(function(tr){
      var isToday = Number(tr.getAttribute('data-day')) === d;
      tr.classList.toggle('today', isToday);
      if (isToday) tr.setAttribute('aria-current','date'); else tr.removeAttribute('aria-current');
    });
  }
  updateStatus();
  setInterval(updateStatus, 60000);

  /* Counters */
  var fmt = new Intl.NumberFormat('en-US');
  function setFinal(el){ el.textContent = fmt.format(Number(el.getAttribute('data-target'))); }
  function animateCounter(el){
    var target = Number(el.getAttribute('data-target')), dur = 1800, start = null;
    function step(ts){
      if (start === null) start = ts;
      var p = Math.min((ts - start)/dur, 1);
      var eased = 1 - Math.pow(1 - p, 3);
      el.textContent = fmt.format(Math.round(target * eased));
      if (p < 1) requestAnimationFrame(step);
    }
    el.textContent = '0';
    requestAnimationFrame(step);
  }
  var counters = document.querySelectorAll('.counter');

  /* Reveal groups: stagger cards within the same grid/list */
  document.querySelectorAll('.grid, .faq-list, .contact-grid').forEach(function(g){
    var i = 0;
    Array.prototype.forEach.call(g.children, function(c){
      if (c.classList.contains('reveal-card')) c.style.setProperty('--i', i++);
    });
  });
  document.querySelectorAll('.reveal-card').forEach(function(c){
    c.addEventListener('transitionend', function(e){
      if (e.propertyName === 'transform' && c.classList.contains('in')) c.classList.add('done');
    });
  });

  var revealEls = document.querySelectorAll('.reveal, .reveal-card');
  if (reduced || !('IntersectionObserver' in window)) {
    revealEls.forEach(function(el){ el.classList.add('in','done'); });
    counters.forEach(setFinal);
  } else {
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(en){
        if (en.isIntersecting) { en.target.classList.add('in'); io.unobserve(en.target); }
      });
    }, {threshold:0.12, rootMargin:'0px 0px -40px 0px'});
    revealEls.forEach(function(el){ io.observe(el); });

    counters.forEach(function(el){ el.textContent = '0'; });

      entries.forEach(function(en){
        if (en.isIntersecting) { animateCounter(en.target); co.unobserve(en.target); }

  }

  /* FAQ */

        item.classList.remove('open');

    window.location.href = href;
  });
