  // day index: 0 = Sunday ... 6 = Saturday; [openMinutes, closeMinutes] or null
  var HOURS = {0:null,1:[480,1080],2:[480,1080],3:[480,1080],4:[480,1080],5:[480,840],6:null};
  function updateStatus(){
    var now = new Date();
    var day = now.getDay();
    var mins = now.getHours()*60 + now.getMinutes();
    var h = HOURS[day];
    var open = !!(h && mins >= h[0] && mins < h[1]);
    document.querySelectorAll('.open-badge').forEach(function(b){
      b.textContent = open ? 'Open now' : 'Closed now';
      b.classList.toggle('is-open', open);
      b.classList.toggle('is-closed', !open);
      b.hidden = false;
    });
    document.querySelectorAll('.hours tr').forEach(function(tr){
      var isToday = Number(tr.getAttribute('data-day')) === day;
      tr.classList.toggle('today', isToday);
      var th = tr.querySelector('th');
      var tag = th.querySelector('.today-tag');
      if (isToday && !tag){
        tag = document.createElement('span');
        tag.className = 'today-tag';
        tag.textContent = 'Today';
        th.appendChild(tag);
      } else if (!isToday && tag){
        tag.remove();
      }
    });
  }
  updateStatus();
  setInterval(updateStatus, 30000);

  /* Counters */

  var counters = Array.prototype.slice.call(document.querySelectorAll('.count'));
  function setFinal(el){ el.textContent = fmt(Number(el.getAttribute('data-target'))); }
  function animateCount(el){
    var target = Number(el.getAttribute('data-target'));
    var duration = 1800, start = null;
    function step(ts){
      if (start === null) start = ts;
      var p = Math.min((ts - start)/duration, 1);
      var eased = 1 - Math.pow(1 - p, 3);
      el.textContent = fmt(target * eased);
      if (p < 1) requestAnimationFrame(step); else setFinal(el);
