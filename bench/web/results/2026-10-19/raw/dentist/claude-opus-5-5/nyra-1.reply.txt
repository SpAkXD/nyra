enum Msg { Toggle(int) }

var open_faq = -1

fn update(m: Msg) {
    match m {
        Msg.Toggle(i) => open_faq = open_faq == i ? -1 : i
    }
}

fn faq_items() -> Node = stack(anim: "stagger") {
    for i, q in faqs {
        card(q.0, "") {

            if i == open_faq {
                p(q.1)
            }
        }
    }
}
