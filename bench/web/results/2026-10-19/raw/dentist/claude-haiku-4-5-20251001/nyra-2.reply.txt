fn isOpenNow() -> bool {
  let hours = [
    (8, 18),
    (8, 18),
    (8, 18),
    (8, 18),
    (8, 14),
    (-1, -1),
    (-1, -1)
  ]
  let now = time.now_ms() / 1000
  let tzOffsetSec = 3600
  let localSec = (now + tzOffsetSec) % 86400
  let hour = localSec / 3600
  let dayOfWeek = ((now + tzOffsetSec) / 86400 + 4) % 7
  let (open, close) = hours[int(dayOfWeek)]
  return open >= 0 && hour >= open && hour < close
}

fn getTodayIndex() -> int {
  let now = time.now_ms() / 1000
  let tzOffsetSec = 3600
  let dayOfWeek = ((now + tzOffsetSec) / 86400 + 4) % 7
  return int(dayOfWeek)
}
