    const menuBtn = document.getElementById('menuBtn');

    // Opening hours and badges
    function updateOpenStatus() {
      const now = new Date();
      const day = now.getDay();
      const hour = now.getHours();
      const minute = now.getMinutes();
      const currentTime = hour * 60 + minute;

      const hours = [
        { open: 8 * 60, close: 18 * 60 }, // Monday
        { open: 8 * 60, close: 18 * 60 }, // Tuesday
        { open: 8 * 60, close: 18 * 60 }, // Wednesday
        { open: 8 * 60, close: 18 * 60 }, // Thursday
        { open: 8 * 60, close: 14 * 60 }, // Friday
        { open: null, close: null }, // Saturday
        { open: null, close: null }  // Sunday
      ];

      const dayHours = hours[day];
      let isOpen = false;

      if (dayHours.open !== null && currentTime >= dayHours.open && currentTime < dayHours.close) {
        isOpen = true;
      }

      const badge = isOpen ? 'Open now' : 'Closed now';

      // Update all badges
      const heroBadge = document.getElementById('heroBadge');
      if (heroBadge) heroBadge.textContent = badge;

      const contactBadge = document.getElementById('contactBadge');
      if (contactBadge) contactBadge.textContent = badge;

      // Highlight today's row
      document.querySelectorAll('.hours-row').forEach(row => {
        row.classList.remove('today');
        if (parseInt(row.dataset.day) === day) {
          row.classList.add('today');
        }
      });
    }

    updateOpenStatus();
    setInterval(updateOpenStatus, 60000);

    // Scroll reveal animations
    function handleScroll() {
      const elements = document.querySelectorAll('.section, .number-item, .service-card, .team-card, .testimonial, .faq-item');

      elements.forEach((el, index) => {
        const rect = el.getBoundingClientRect();
        if (rect.top < window.innerHeight * 0.8) {
          el.classList.add('appear');
        }
      });
    }

    window.addEventListener('scroll', handleScroll);
    handleScroll();

    // Number counter animation
    function animateNumbers() {
      const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      const numberItems = document.querySelectorAll('.number-item');

      numberItems.forEach(item => {
        const rect = item.getBoundingClientRect();
        if (rect.top < window.innerHeight * 0.8) {
          const target = parseInt(item.dataset.target);
          const valueEl = item.querySelector('.number-value');

          if (prefersReducedMotion) {
            valueEl.textContent = target.toLocaleString();
          } else {
            const current = parseInt(valueEl.textContent) || 0;
            if (current < target) {
              const increment = Math.ceil(target / 50);
              const newValue = Math.min(current + increment, target);
              valueEl.textContent = newValue.toLocaleString();
            }
          }
        }
      });
    }

    setInterval(animateNumbers, 30);

    // Booking form
