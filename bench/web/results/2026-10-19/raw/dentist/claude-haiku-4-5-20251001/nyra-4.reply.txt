enum Msg { 
  ToggleFaq(int)
  SubmitBooking
}

var faqOpen: [bool] = [false, false, false, false, false]
var bookingData: [str: str] = [:]

fn update(m: Msg) {
  match m {
    Msg.ToggleFaq(i) => faqOpen[i] = !faqOpen[i]
    Msg.SubmitBooking => {
      let name = bookingData.get("name", "")
      let phone = bookingData.get("phone", "")
      let email = bookingData.get("email", "")
      let treatment = bookingData.get("treatment", "")
      let date = bookingData.get("date", "")
      let message = bookingData.get("message", "")
      let body = "Name: {name}\nPhone: {phone}\nEmail: {email}\nTreatment: {treatment}\nDate: {date}\nMessage: {message}"
      print("mailto:recepce@zelenadental.cz?subject=Appointment%20request&body={body}")
    }
  }
}

fn getOpenStatus() -> str = {
  let now = time.now_ms()
  let hours = [
    (1, 8, 18),

    (3, 8, 18),
    (4, 8, 18),
    (5, 8, 14),
    (6, 0, 0),
    (0, 0, 0)
  ]
  "Open now"
}

fn serviceIcon(name: str) -> str = {
  match name {

    "Fillings" => ":shield:"
    "Whitening" => ":smile:"
    "Children's dentistry" => ":heart:"

    _ => ":star:"
  }
}
