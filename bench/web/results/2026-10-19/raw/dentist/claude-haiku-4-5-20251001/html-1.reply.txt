    // Opening hours and badge
    function updateOpenStatus() {
      const hours = {
        0: { name: 'Sunday', times: [] },
        1: { name: 'Monday', times: [8, 18] },
        2: { name: 'Tuesday', times: [8, 18] },
        3: { name: 'Wednesday', times: [8, 18] },
        4: { name: 'Thursday', times: [8, 18] },
        5: { name: 'Friday', times: [8, 14] },
        6: { name: 'Saturday', times: [] }
      };

      const now = new Date();
      const day = now.getDay();
      const hour = now.getHours();
      const minute = now.getMinutes();

      const dayHours = hours[day];
      let isOpen = false;

      if (dayHours.times.length > 0) {
        const [openHour, closeHour] = dayHours.times;
        const currentMinutes = hour * 60 + minute;
        const openMinutes = openHour * 60;
        const closeMinutes = closeHour * 60;
        isOpen = currentMinutes >= openMinutes && currentMinutes < closeMinutes;
      }

      // Update badge
      const badge = document.getElementById('heroBadge');
      if (badge) {
        badge.textContent = isOpen ? 'Open now' : 'Closed now';
      }

      // Highlight today's row
      const dayRowIds = ['sun', 'mon', 'tue', 'wed', 'thu', 'fri', 'sat'];
      const rows = document.querySelectorAll('.hours-table tr');
      rows.forEach(row => row.classList.remove('today'));

      const todayRow = document.getElementById(dayRowIds[day]);
      if (todayRow) {
        todayRow.classList.add('today');
        // Add badge to today's row
        let badge = todayRow.querySelector('.today-badge');
        if (!badge) {
          badge = document.createElement('span');
          badge.className = 'today-badge';
          badge.style.marginLeft = '1rem';
          badge.style.background = isOpen ? '#0F766E' : '#666';
          badge.style.color = 'white';
          badge.style.padding = '4px 12px';
          badge.style.borderRadius = '12px';
          badge.style.fontSize = '0.75rem';
          badge.style.fontWeight = '600';
          badge.textContent = isOpen ? 'Open now' : 'Closed now';
          const timeCell = todayRow.querySelector('td:last-child');
          timeCell.appendChild(badge);
        }
      }
    }

    updateOpenStatus();
    setInterval(updateOpenStatus, 60000);

    // Number counter animation
    function animateNumbers() {
      const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
      const numbers = document.querySelectorAll('.number-item h3');

      numbers.forEach(element => {
        const target = parseInt(element.dataset.target);
        
        if (prefersReducedMotion) {
          element.textContent = target.toLocaleString('cs-CZ');
        } else {
          let current = 0;
          const duration = 2000;
          const start = Date.now();

          function update() {
            const elapsed = Date.now() - start;
            const progress = Math.min(elapsed / duration, 1);
            current = Math.floor(target * progress);
            element.textContent = current.toLocaleString('cs-CZ');

            if (progress < 1) {
              requestAnimationFrame(update);
            } else {
              element.textContent = target.toLocaleString('cs-CZ');
            }
          }

          update();
        }
      });
    }

    // Intersection observer for animations
    const observerOptions = {
      threshold: 0.1,
      rootMargin: '0px 0px -50px 0px'
    };

    const observer = new IntersectionObserver(function(entries) {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          if (entry.target.classList.contains('numbers-section')) {
            animateNumbers();
            observer.unobserve(entry.target);
          }
        }
      });
    }, observerOptions);

    observer.observe(document.querySelector('.numbers-section'));

    // FAQ toggle
    function toggleFAQ(button) {

      const toggle = button.querySelector('.faq-toggle');
      const isOpen = answer.classList.contains('open');

      document.querySelectorAll('.faq-answer').forEach(a => a.classList.remove('open'));
      document.querySelectorAll('.faq-toggle').forEach(t => t.textContent = '+');

      if (!isOpen) {
        answer.classList.add('open');
        toggle.textContent = '−';
      }
    }
