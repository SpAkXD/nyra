fn get_day_name(day: int) -> str = match day {
  0 => "Monday", 1 => "Tuesday", 2 => "Wednesday", 3 => "Thursday", 4 => "Friday", 5 => "Saturday", 6 => "Sunday", _ => ""
}

fn is_open_now() -> bool {
  let now = time.now_ms()
  let hours_since_epoch = (now / 3600000) % 24
  let day_index = (now / 86400000) % 7
  
  if day_index == 5 || day_index == 6 {
    return false
  }
  
  if day_index == 4 {
    return hours_since_epoch >= 8 && hours_since_epoch < 14
  }
  
  return hours_since_epoch >= 8 && hours_since_epoch < 18
}

enum Msg { ToggleFaq(int) }

fn update(m: Msg) { 
  match m { 
    Msg.ToggleFaq(i) => {
      faq_open[i] = !faq_open[i]
    }
  } 
}

fn faq_section() -> Node = stack {
  for i, q in faq_items {
    stack(cls: "faq-item", hover: "lift") {

      if faq_open[i] {
        p(q.1)
      }
    }
  }
}
